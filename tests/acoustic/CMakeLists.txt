cmake_minimum_required(VERSION 3.16)
project(life_acoustic_host LANGUAGES C CXX)
set(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(GGWAVE_SUPPORT_SDL2 OFF CACHE BOOL "" FORCE)
set(GGWAVE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGWAVE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP_ROOT}/components/ggwave/upstream" "${CMAKE_BINARY_DIR}/ggwave")
find_package(ZLIB REQUIRED)
add_executable(acoustic_host acoustic_host.cpp "${APP_ROOT}/main/life_audio_protocol.c" "${APP_ROOT}/main/life.c")
target_include_directories(acoustic_host PRIVATE "${APP_ROOT}/main")
target_link_libraries(acoustic_host PRIVATE ggwave ZLIB::ZLIB)
target_compile_features(acoustic_host PRIVATE cxx_std_11)
target_compile_options(acoustic_host PRIVATE -Wall -Wextra -Werror -UNDEBUG)
